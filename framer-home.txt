<script>
(function () {
  var f = document.getElementById("informativo");
  window.addEventListener("message", function (e) {
    if (!e.data || e.data.tipo !== "informativo:altura") return;
    var h = parseInt(e.data.altura, 10);
    if (h > 200) f.style.height = h + "px";
  });
})();
</script>
